async function handleLogin(e) { e.preventDefault(); const u = document.getElementById('login-user').value.trim(); const p = document.getElementById('login-pass').value.trim(); try { const { data, error } = await supabaseClient .from('users') .select('*') .eq('username', u) .eq('password', p); if (error) { alert("Erro do Supabase: " + error.message); return; } if (!data || data.length === 0) { alert("Utilizador ou senha incorretos! Verifique se a tabela 'users' tem dados."); return; } currentUser = data[0]; localStorage.setItem('reg_current_user', JSON.stringify(currentUser)); document.getElementById('login-screen').classList.add('hidden'); document.getElementById('current-user-display').innerText = currentUser.username; document.getElementById('current-role-display').innerText = currentUser.role === 'admin' ? 'Administrador Mestre' : 'Técnico'; await logAuditAction("Login", `Utilizador ${currentUser.username} entrou no sistema.`); await loadDataFromCloud(); } catch (err) { alert("Erro inesperado no login: " + err.message); } }
